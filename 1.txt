    function renderQuestions(){
      questionsEl.innerHTML = '';
      qa.forEach((item, idx) => {
        const div = document.createElement('div');
        div.className = 'q';
        div.dataset.idx = idx;
        div.style.display = idx===current ? 'block' : 'none';
        const h = document.createElement('h3');
        h.textContent = `${idx+1}. ${item.q}`;
        div.appendChild(h);
        const opts = document.createElement('div');
        opts.className = 'opts';
        item.opts.forEach((op, oi) => {
          const b = document.createElement('button');
          b.type = 'button';
          b.className = 'opt';
          b.textContent = op.t;
          b.addEventListener('click', ()=>{
            selectOption(idx, oi);
          });
          if (answers[idx] === oi) b.classList.add('active');
          opts.appendChild(b);
        });
        div.appendChild(opts);
        questionsEl.appendChild(div);
      });
      updateProgress();
    }

    function selectOption(qidx, optidx){
      answers[qidx] = optidx;
      // 更新 UI
      const qdiv = document.querySelector(`.q[data-idx="${qidx}"]`);
      qdiv.querySelectorAll('.opt').forEach((el,i)=> el.classList.toggle('active', i===optidx));
      updateProgress();
      // 自动下一题（如果不是最后一题）
      if (qidx < qa.length-1) {
        current = qidx+1;
        showCurrent();
      }
    }

    function showCurrent(){
      document.querySelectorAll('.q').forEach(div=>{
        div.style.display = div.dataset.idx == current ? 'block' : 'none';
      });
      updateProgress();
    }

    function updateProgress(){
      const done = answers.filter(a => a !== null).length;
      const prog = Math.round(done / qa.length * 100);
      document.getElementById('prog').style.width = prog + '%';
      document.getElementById('qcount').textContent = `已答 ${done} / ${qa.length}`;
      // 控制按钮显示
      document.getElementById('prev').style.display = current===0 ? 'none' : '';
      document.getElementById('next').style.display = current===qa.length-1 ? 'none' : '';
      document.getElementById('submit').style.display = done===qa.length ? '' : 'none';
    }

    document.getElementById('prev').addEventListener('click', ()=>{
      if (current>0) current--;
      showCurrent();
    });
    document.getElementById('next').addEventListener('click', ()=>{
      if (current < qa.length-1) current++;
      showCurrent();
